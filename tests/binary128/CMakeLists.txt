cmake_minimum_required(VERSION 3.24)
project(gesher_binary128_contracts LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(AOSP_DIR "" CACHE PATH "Pinned android-16.0.0_r4 AOSP checkout")
if(NOT AOSP_DIR)
  message(FATAL_ERROR "Pass -DAOSP_DIR=<pinned AOSP checkout>")
endif()
include(../../compat/bionic/binary128_parse.cmake)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/pinned_parsefloat.inc
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare_prefix_reference.py
          ${CMAKE_CURRENT_BINARY_DIR}/pinned_parsefloat.inc
  DEPENDS prepare_prefix_reference.py
          ../../probes/native3/wide/upstream/stdio/parsefloat.c)
add_executable(binary128_prefix prefix_contracts.cc ${CMAKE_CURRENT_BINARY_DIR}/pinned_parsefloat.inc)
target_include_directories(binary128_prefix PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(binary128_prefix PRIVATE gesher_bionic_binary128)
target_compile_options(binary128_prefix PRIVATE -Wall -Wextra -Werror)
add_executable(binary128_records backend_records.cc)
target_link_libraries(binary128_records PRIVATE gesher_bionic_binary128)
target_compile_options(binary128_records PRIVATE -Wall -Wextra -Werror)
add_executable(binary128_concurrency concurrency_contracts.cc)
target_link_libraries(binary128_concurrency PRIVATE gesher_bionic_binary128)
target_compile_options(binary128_concurrency PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  add_executable(arm64_numeric_fenv_test ../native/arm64_numeric_fenv_test.cc)
  target_compile_options(arm64_numeric_fenv_test PRIVATE -Wall -Wextra -Werror
    -frounding-math $<$<CXX_COMPILER_ID:GNU>:-Wno-unknown-pragmas>)
  if(WIN32)
    target_link_options(arm64_numeric_fenv_test PRIVATE -static)
  endif()
endif()
if(WIN32)
  target_link_options(binary128_records PRIVATE -static)
  target_link_options(binary128_prefix PRIVATE -static)
  target_link_options(binary128_concurrency PRIVATE -static)
endif()
