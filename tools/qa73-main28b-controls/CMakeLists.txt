cmake_minimum_required(VERSION 3.24)
project(qa73_main_controls LANGUAGES C CXX ASM)
if(NOT WIN32 OR NOT QA73_SOURCE_ROOT)
  message(FATAL_ERROR "A verified Windows production source root is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_C_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${QA73_SOURCE_ROOT}/tools/pipe-component" component)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/diagnostic.py"
  generate --source "${QA73_SOURCE_ROOT}" --out "${CMAKE_BINARY_DIR}/generated"
  COMMAND_ERROR_IS_FATAL ANY)
add_executable(native_child_observed "${CMAKE_BINARY_DIR}/generated/ofd-native-child.cc")
target_compile_definitions(native_child_observed PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(native_child_observed PRIVATE -Wall -Wextra -Werror)
target_link_options(native_child_observed PRIVATE -static)
add_executable(empty_stdio_options empty_stdio_options.cc)
target_compile_options(empty_stdio_options PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(empty_stdio_options PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
  _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_options(empty_stdio_options PRIVATE -static -municode)
target_link_libraries(empty_stdio_options PRIVATE gesher_exec_spawn)
