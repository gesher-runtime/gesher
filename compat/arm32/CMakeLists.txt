# Standalone ARM32 components. Deliberately not linked into dalvikvm or packaged
# as an available app ABI until the Android JNI/Bionic bridge is implemented.
cmake_minimum_required(VERSION 3.24)
project(gesher_arm32_components LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
include(CTest)

add_library(gesher_arm32_core STATIC
  arm32_memory.cc arm32_abi.cc arm32_elf.cc arm32_image_memory.cc)
target_include_directories(gesher_arm32_core PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include)
target_link_libraries(gesher_arm32_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(gesher_arm32_core PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(gesher_arm32_core PRIVATE -Wall -Wextra -Werror -pedantic -fexceptions)
endif()

if(BUILD_TESTING)
  foreach(component memory abi elf)
    add_executable(arm32_${component}_test ../../tests/arm32/${component}_test.cc)
    target_link_libraries(arm32_${component}_test PRIVATE gesher_arm32_core)
    # memory_test overrides global new to inject a throwing bad_alloc.
    # MSVC needs C++ exception unwinding enabled in the caller too;
    # otherwise a deliberate allocation fault terminates with 0xC0000409.
    if(MSVC)
      target_compile_options(arm32_${component}_test PRIVATE /EHsc)
    endif()
    add_test(NAME arm32_${component} COMMAND arm32_${component}_test)
  endforeach()
endif()

option(GESHER_ARM32_WITH_DYNARMIC "Build and test the experimental A32 executor" OFF)
if(GESHER_ARM32_WITH_DYNARMIC)
  include(CheckCXXSourceCompiles)
  check_cxx_source_compiles("\n#if !defined(__x86_64__) && !defined(_M_X64)\n#error The checked A32 executor currently requires an x86-64 host backend\n#endif\nstatic_assert(sizeof(void*) == 8);\nint main() { return 0; }" ARM32_X64_BACKEND)
  if(NOT ARM32_X64_BACKEND)
    message(FATAL_ERROR "The experimental A32 executor supports x86-64 hosts only; core tests remain portable")
  endif()
  if(NOT EXISTS "${DYNARMIC_DIR}/src/dynarmic/interface/A32/a32.h")
    message(FATAL_ERROR "Set DYNARMIC_DIR to the pinned Dynarmic checkout (docs/arm32.md)")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/patch_dynarmic.py ${DYNARMIC_DIR} --check
    RESULT_VARIABLE ARM32_PATCH_STATUS OUTPUT_VARIABLE ARM32_PATCH_OUTPUT ERROR_VARIABLE ARM32_PATCH_ERROR)
  if(NOT ARM32_PATCH_STATUS EQUAL 0)
    message(FATAL_ERROR "Apply the checked A32 patch first: python3 compat/arm32/patch_dynarmic.py <DYNARMIC_DIR>\n${ARM32_PATCH_OUTPUT}${ARM32_PATCH_ERROR}")
  endif()
  set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
  set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
  set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
  set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
  set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  # Dynarmic's externals change BUILD_TESTING; retain our caller's choice.
  set(ARM32_BUILD_TESTING ${BUILD_TESTING})
  add_subdirectory(${DYNARMIC_DIR} dynarmic-build EXCLUDE_FROM_ALL)
  add_library(gesher_arm32_executor STATIC arm32_executor.cc)
  target_link_libraries(gesher_arm32_executor PUBLIC gesher_arm32_core PRIVATE dynarmic)
  if(MSVC)
    target_compile_options(gesher_arm32_executor PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(gesher_arm32_executor PRIVATE -Wall -Wextra -Werror -pedantic -fexceptions)
  endif()
  add_library(gesher_arm32_program STATIC arm32_program.cc)
  target_link_libraries(gesher_arm32_program PUBLIC gesher_arm32_executor)
  add_library(gesher_arm32_bionic STATIC arm32_bionic.cc)
  target_link_libraries(gesher_arm32_bionic PUBLIC gesher_arm32_program)

  # A host JNIEnv is required for the JNI32 bridge. Only jni.h types are
  # needed: the tests supply a fake JNIEnv and never link a JVM.
  find_package(JNI REQUIRED)
  add_library(gesher_arm32_jni STATIC arm32_jni.cc)
  target_include_directories(gesher_arm32_jni PUBLIC ${JNI_INCLUDE_DIRS})
  target_link_libraries(gesher_arm32_jni PUBLIC gesher_arm32_program)
  if(MSVC)
    target_compile_options(gesher_arm32_program PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(gesher_arm32_program PRIVATE -Wall -Wextra -Werror -pedantic -fexceptions)
  endif()
  if(MSVC)
    target_compile_options(gesher_arm32_bionic PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(gesher_arm32_bionic PRIVATE -Wall -Wextra -Werror -pedantic -fexceptions)
  endif()
  if(MSVC)
    target_compile_options(gesher_arm32_jni PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(gesher_arm32_jni PRIVATE -Wall -Wextra -Werror -pedantic -fexceptions)
  endif()
  if(ARM32_BUILD_TESTING)
    add_executable(arm32_jni_test ../../tests/arm32/jni_test.cc)
    target_link_libraries(arm32_jni_test PRIVATE gesher_arm32_jni)
    add_executable(arm32_dependencies_test ../../tests/arm32/dependencies_test.cc)
    target_link_libraries(arm32_dependencies_test PRIVATE gesher_arm32_program)
    add_executable(arm32_bionic_test ../../tests/arm32/bionic_test.cc)
    target_link_libraries(arm32_bionic_test PRIVATE gesher_arm32_bionic)
    add_executable(arm32_program_test ../../tests/arm32/program_test.cc)
    target_link_libraries(arm32_program_test PRIVATE gesher_arm32_program)
    add_executable(arm32_executor_test ../../tests/arm32/executor_test.cc)
    target_link_libraries(arm32_executor_test PRIVATE gesher_arm32_executor)
    if(MSVC)
      foreach(target IN ITEMS arm32_jni_test arm32_dependencies_test
              arm32_bionic_test arm32_program_test arm32_executor_test)
        target_compile_options(${target} PRIVATE /EHsc)
      endforeach()
    endif()
    add_test(NAME arm32_executor COMMAND arm32_executor_test)
    set_tests_properties(arm32_executor PROPERTIES TIMEOUT 60)
    set(ARM32_GUEST_FIXTURE "" CACHE FILEPATH "Compiled ARM EABI5 softfp ELF32 fixture")
    if(ARM32_GUEST_FIXTURE)
      add_test(NAME arm32_compiled_elf COMMAND arm32_executor_test ${ARM32_GUEST_FIXTURE})
      set_tests_properties(arm32_compiled_elf PROPERTIES TIMEOUT 60)
    endif()
    set(ARM32_GUEST_FIXTURES "" CACHE STRING "Semicolon-separated compiled ARM/Thumb ELF32 fixtures")
    foreach(fixture IN LISTS ARM32_GUEST_FIXTURES)
      get_filename_component(fixture_name ${fixture} NAME_WE)
      add_test(NAME arm32_${fixture_name} COMMAND arm32_executor_test ${fixture})
      set_tests_properties(arm32_${fixture_name} PROPERTIES TIMEOUT 60)
      add_test(NAME arm32_program_${fixture_name} COMMAND arm32_program_test ${fixture})
      set_tests_properties(arm32_program_${fixture_name} PROPERTIES TIMEOUT 60)
    endforeach()
    set(ARM32_ALIGNED_FIXTURE_DIR "" CACHE PATH "Directory containing 64-KiB-aligned real ARM/Thumb fixtures")
    if(ARM32_ALIGNED_FIXTURE_DIR)
      foreach(mode IN ITEMS arm thumb)
        add_test(NAME arm32_alignment_${mode}
          COMMAND arm32_program_test
            "${ARM32_ALIGNED_FIXTURE_DIR}/fixture-${mode}.so"
            "${ARM32_ALIGNED_FIXTURE_DIR}/fixture-${mode}-align64k.so")
        set_tests_properties(arm32_alignment_${mode} PROPERTIES TIMEOUT 60)
      endforeach()
    endif()
    set(ARM32_GUEST_DEPENDENCY_DIR "" CACHE PATH "Directory containing two-DSO ARM/Thumb fixtures")
    if(ARM32_GUEST_DEPENDENCY_DIR)
      foreach(mode IN ITEMS arm thumb)
        add_test(NAME arm32_dependencies_${mode}
          COMMAND arm32_dependencies_test
            "${ARM32_GUEST_DEPENDENCY_DIR}/consumer-${mode}.so"
            "${ARM32_GUEST_DEPENDENCY_DIR}/dependency-${mode}.so"
            "consumer-${mode}.so" "dependency-${mode}.so")
        set_tests_properties(arm32_dependencies_${mode} PROPERTIES TIMEOUT 60)
      endforeach()
    endif()
    set(ARM32_GUEST_JNI_FIXTURES "" CACHE STRING "Semicolon-separated compiled ARM/Thumb JNI32 fixture paths")
    foreach(fixture IN LISTS ARM32_GUEST_JNI_FIXTURES)
      get_filename_component(fixture_name ${fixture} NAME_WE)
      add_test(NAME arm32_jni_${fixture_name} COMMAND arm32_jni_test ${fixture})
      set_tests_properties(arm32_jni_${fixture_name} PROPERTIES TIMEOUT 60)
    endforeach()
    set(ARM32_GUEST_BIONIC_FIXTURES "" CACHE STRING "Semicolon-separated compiled ARM/Thumb Bionic32 fixture paths")
    foreach(fixture IN LISTS ARM32_GUEST_BIONIC_FIXTURES)
      get_filename_component(fixture_name ${fixture} NAME_WE)
      add_test(NAME arm32_bionic_${fixture_name} COMMAND arm32_bionic_test ${fixture})
      set_tests_properties(arm32_bionic_${fixture_name} PROPERTIES TIMEOUT 60)
    endforeach()
  endif()
endif()
