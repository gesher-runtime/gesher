# Framework references to gesher-app.jar classes that app/gen/BootLinkCheck.java may
# leave unresolved, as owner.name:descriptor. Anything else unresolved fails
# app/build.sh: a NoSuchMethodError or NoSuchFieldError waiting for the framework path
# that uses it, as issue #303's BluetoothDevice.prepareToEnterProcess was. Each entry
# says why it is accepted; remove an entry the check reports as stale.

## Paths Gesher does not run

# ActivityThread.initializeMainlineModules, which only ActivityThread.main calls.
# Gesher's processes start in GesherAppProcess and GesherAppDriver instead (they
# construct and attach ActivityThread themselves); the initializers'
# registerServiceWrappers still run from SystemServiceRegistry's static initializer.
android.bluetooth.BluetoothFrameworkInitializer.setBinderCallsStatsInitializer:(Ljava/util/function/Consumer;)V
android.bluetooth.BluetoothFrameworkInitializer.setBluetoothServiceManager:(Landroid/os/BluetoothServiceManager;)V
android.media.MediaFrameworkInitializer.setMediaServiceManager:(Landroid/media/MediaServiceManager;)V
android.nfc.NfcFrameworkInitializer.setNfcServiceManager:(Landroid/nfc/NfcServiceManager;)V

android.os.StatsFrameworkInitializer.setStatsServiceManager:(Landroid/os/StatsServiceManager;)V

# ZygoteInit.preload: Gesher has no zygote.
android.net.http.HttpEngine.preload:()V

## Known gaps (reachable; tracked for follow-up)

# SystemServiceRegistry's TETHERING_SERVICE fetcher constructs TetheringManager(Context,
# Supplier<IBinder>). app/compat-java's TetheringManager is a placeholder that only lets
# SystemServiceRegistry's static initializer name the class, and Gesher has no tethering
# service: an app's getSystemService(TETHERING_SERVICE) (or Context.getSystemService
# (TetheringManager.class)) fails with NoSuchMethodError instead of answering a manager.
# A faithful fix needs the Connectivity module's TetheringManager over a tethering
# service that reports tethering unsupported.
android.net.TetheringManager.<init>:(Landroid/content/Context;Ljava/util/function/Supplier;)V
